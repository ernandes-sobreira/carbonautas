function fileExt(name,url=''){
  const source=(name||'').split('?')[0] || (url||'').split('?')[0];
  const m=source.toLowerCase().match(/\.([a-z0-9]+)$/);
  if(m)return m[1];
  try{
    const u=new URL(url);
    const mm=u.pathname.toLowerCase().match(/\.([a-z0-9]+)$/);
    return mm?mm[1]:'';
  }catch{return ''}
}

function extFromMime(mt=''){
  mt=String(mt).toLowerCase().split(';')[0].trim();
  if(mt.includes('wordprocessingml'))return 'docx';
  if(mt.includes('spreadsheetml'))return 'xlsx';
  if(mt.includes('presentationml'))return 'pptx';
  return '';
}
function filenameFromDisposition(v=''){
  const star=String(v).match(/filename\*=UTF-8''([^;]+)/i);if(star){try{return decodeURIComponent(star[1].replace(/^"|"$/g,''))}catch{}}
  const m=String(v).match(/filename="?([^";]+)"?/i);return m?m[1].trim():'';
}
async function resolveOfficeFile(p){
  let ext=String(p.detectedExt||p.officeExt||'').toLowerCase();
  let fileName=p.fileName||'';
  if(!ext)ext=fileExt(fileName||p.titulo||'',p.url||'');
  if(!ext)ext=extFromMime(p.mimeType||p.mime||p.contentType||'');
  if((!ext||!fileName)&&/^https?:\/\//i.test(p.url||'')){
    try{
      const r=await fetch(p.url,{method:'HEAD',redirect:'follow'});
      if(r.ok){
        if(!ext)ext=extFromMime(r.headers.get('content-type')||'');
        if(!fileName)fileName=filenameFromDisposition(r.headers.get('content-disposition')||'');
      }
    }catch(e){console.warn('PROBE HEAD:',e.message)}
  }
  // Alguns serviços não respondem HEAD corretamente; um GET mínimo ainda revela Content-Type.
  if(!ext&&/^https?:\/\//i.test(p.url||'')){
    try{
      const r=await fetch(p.url,{headers:{Range:'bytes=0-31'},redirect:'follow'});
      if(r.ok||r.status===206)ext=extFromMime(r.headers.get('content-type')||'');
      try{r.body?.cancel?.()}catch{}
    }catch(e){console.warn('PROBE GET:',e.message)}
  }
  return {ext,fileName:fileName||p.titulo||('arquivo'+(ext?'.'+ext:''))};
}

async function ensureTemplate(ext){
  if(!SUPPORTED.has(ext))throw new Error('Formato de modelo não suportado.');
  const out=path.join(TEMPLATES,'new.'+ext);
  if(fs.existsSync(out)&&fs.statSync(out).size>1000)return out;
  const r=await fetch(TEMPLATE_URLS[ext]);
  if(!r.ok)throw new Error('Não consegui obter o modelo vazio: HTTP '+r.status);
  fs.writeFileSync(out,Buffer.from(await r.arrayBuffer()),{mode:0o600});
  return out;
}

async function getLabPublication(pubId,user){
  const snap=await db.collection('rede_publicacoes').doc(pubId).get();
  if(!snap.exists){const e=new Error('Base de dados não encontrada.');e.status=404;throw e;}
  const p={...snap.data(),id:snap.id};
  if(!labAccess(p,user)){const e=new Error('Você não tem acesso a esta base.');e.status=403;throw e;}
  if(p.tipo!=='arquivo'){const e=new Error('O item selecionado não é um arquivo de dados.');e.status=400;throw e;}
  const ext=fileExt(p.fileName||p.titulo||'',p.url||'');
  if(!LAB_EXTS.has(ext)){const e=new Error('O Laboratório aceita CSV e Excel (XLSX/XLS).');e.status=400;throw e;}
  if(!/^https?:\/\//i.test(p.url||'')){const e=new Error('A base não possui URL válida.');e.status=400;throw e;}
  return {...p,ext};
}
function safeDatasetUrl(raw){
  const u=new URL(raw);
  const ok=u.hostname==='firebasestorage.googleapis.com' || u.hostname==='storage.googleapis.com';
  if(!ok)throw new Error('Por segurança, o Laboratório processa arquivos armazenados no Firebase Storage.');
  return u.toString();
}
async function downloadLabDataset(p,target){
  const r=await fetch(safeDatasetUrl(p.url));
  if(!r.ok)throw new Error('Não consegui baixar a base do Firebase: HTTP '+r.status);
  const size=Number(r.headers.get('content-length')||0);
  if(size>50*1024*1024)throw new Error('Base maior que 50 MB. Divida ou filtre antes desta versão do Laboratório.');
  const buf=Buffer.from(await r.arrayBuffer());
  if(buf.length>50*1024*1024)throw new Error('Base maior que 50 MB.');
  fs.writeFileSync(target,buf,{mode:0o600});
}
function runLabPython(payload,timeoutMs=150000){
  return new Promise((resolve,reject)=>{
    const cp=spawn(LAB_PYTHON,[LAB_ENGINE],{cwd:ROOT,stdio:['pipe','pipe','pipe']});
    let out='',err='';
    const timer=setTimeout(()=>{cp.kill('SIGKILL');reject(new Error('A análise ultrapassou o tempo máximo. Reduza a base ou simplifique o modelo.'));},timeoutMs);
    cp.stdout.on('data',d=>{out+=d.toString();if(out.length>20*1024*1024){cp.kill('SIGKILL');}});
    cp.stderr.on('data',d=>{err+=d.toString();if(err.length>4*1024*1024)err=err.slice(-4*1024*1024);});
    cp.on('error',e=>{clearTimeout(timer);reject(e);});
    cp.on('close',code=>{
      clearTimeout(timer);
      let j;try{j=JSON.parse(out.trim());}catch{j=null;}
      if(code!==0 || !j?.ok){console.error('LAB PY:',err.slice(-2500));return reject(new Error(j?.error||'Falha no motor estatístico.'));}
      resolve(j);
    });
    cp.stdin.end(JSON.stringify(payload));
  });
}

function normalizeLabContext(raw,p,user){
  const allowed=new Set(['project','article','product','independent','exploratory']);
  const type=allowed.has(String(raw?.labContextType||raw?.type||'')) ? String(raw.labContextType||raw.type) : (p.labContextType||'project');
  const memberId= type==='project' ? String(raw?.labMemberId||raw?.memberId||p.labMemberId||p.memberId||user.memberId||'') : String(p.memberId||user.memberId||'');
  const memberName=String(raw?.labMemberNome||raw?.memberName||p.labMemberNome||p.memberNome||user.nome||'');
  const project=String(raw?.labProject||raw?.project||p.labProject||'').trim();
  const chapter= type==='project' ? String(raw?.labChapter||raw?.chapter||p.labChapter||'').trim() : '';
  if(type==='project'){
    if(!memberId) throw Object.assign(new Error('Selecione o aluno/responsável do projeto.'),{status:400});
    if(!project) throw Object.assign(new Error('Informe o projeto ao qual esta base pertence.'),{status:400});
    if(!chapter) throw Object.assign(new Error('Informe o capítulo/etapa ao qual esta base pertence.'),{status:400});
  } else if(['article','product','independent'].includes(type) && !project){
    throw Object.assign(new Error('Informe o nome do artigo, produto ou pesquisa.'),{status:400});
  }
  return {labContextType:type,labMemberId:memberId,labMemberNome:memberName,labProject:project,labChapter:chapter};
}

    r['metrics']=[{'label':'N','value':int(model.nobs)},{'label':'R²','value':round(float(model.rsquared),4)},{'label':'R² ajustado','value':round(float(model.rsquared_adj),4)},{'label':'AIC','value':round(float(model.aic),2)},{'label':'BIC','value':round(float(model.bic),2)}]
    ci=model.conf_int(); rows=[]
    for term in model.params.index:
        rows.append({'Termo':term,'Estimativa':model.params[term],'Erro padrão':model.bse[term],'t':model.tvalues[term],'p':model.pvalues[term],'IC95% inferior':ci.loc[term,0],'IC95% superior':ci.loc[term,1]})
    r['tables'].append(write_table(pd.DataFrame(rows),outdir,'regressao_coeficientes','Coeficientes do modelo'))
    resid=model.resid; fitted=model.fittedvalues
    sh_p=np.nan
    if 3<=len(resid)<=5000:
        sh_p=stats.shapiro(resid).pvalue
        if sh_p<.05:r['warnings'].append(f'Resíduos desviam da normalidade (Shapiro-Wilk p={fmt_p(sh_p)}). Avalie QQ-plot e robustez.')
    bp=het_breuschpagan(resid,model.model.exog); bp_p=bp[1]
    if bp_p<.05:r['warnings'].append(f'Indício de heterocedasticidade (Breusch-Pagan p={fmt_p(bp_p)}). Considere erros-padrão robustos/transformação.')
    dw=float(durbin_watson(resid));
    if dw<1.5 or dw>2.5:r['warnings'].append(f'Durbin-Watson={dw:.2f}; pode haver autocorrelação residual se os dados forem ordenados/temporais.')
    diag=pd.DataFrame([{'Shapiro p':sh_p,'Breusch-Pagan p':bp_p,'Durbin-Watson':dw}])
    if len(xs)>1:
        vx=sm.add_constant(d[xs],has_constant='add')
        vifs=[]
        for i,c in enumerate(vx.columns):
            if c=='const': continue
            vifs.append({'Variável':c,'VIF':variance_inflation_factor(vx.values,i)})
        vdf=pd.DataFrame(vifs); r['tables'].append(write_table(vdf,outdir,'regressao_vif','Colinearidade (VIF)'))
        if (vdf['VIF']>5).any(): r['warnings'].append('Há VIF > 5, sugerindo colinearidade relevante entre preditores.')
    r['tables'].append(write_table(diag,outdir,'regressao_diagnostico','Diagnósticos do modelo'))
    if len(xs)==1:
        x=xs[0];fig,ax=plt.subplots(figsize=(4.1,3.3));ax.scatter(d[x],d[y],s=18,alpha=.65,color=PALETTE[0],edgecolors='white',linewidths=.25)
        xx=np.linspace(d[x].min(),d[x].max(),150);Xp=sm.add_constant(pd.DataFrame({x:xx}),has_constant='add');pred=model.get_prediction(Xp).summary_frame(alpha=.05)
        ax.plot(xx,pred['mean'],color=PALETTE[1],lw=1.6);ax.fill_between(xx,pred['mean_ci_lower'].to_numpy(),pred['mean_ci_upper'].to_numpy(),color=PALETTE[1],alpha=.14,lw=0)
        ax.set_xlabel(x);ax.set_ylabel(y);ax.spines[['top','right']].set_visible(False);ax.set_title(f'{y} ~ {x}\nR²={model.rsquared:.3f}, p={fmt_p(model.f_pvalue)}')
        r['figures'].append(save_figure(fig,outdir,'fig_regressao','Relação observada, ajuste e IC95%'))
    fig,axs=plt.subplots(1,2,figsize=(7,3.1));axs[0].scatter(fitted,resid,s=14,alpha=.6,color=PALETTE[0]);axs[0].axhline(0,color='black',lw=.8);axs[0].set_xlabel('Valores ajustados');axs[0].set_ylabel('Resíduos');axs[0].spines[['top','right']].set_visible(False)

    r['figures'].append(save_figure(fig,outdir,'fig_regressao_diagnosticos','Diagnósticos visuais do modelo'))
    return r


def analysis_group_test(df,opt,outdir,kind):
    y=opt.get('y');g=opt.get('group');ensure_cols(df,[y,g]);d=df[[y,g]].dropna();
    if y not in numeric_cols(d):raise ValueError('A variável resposta deve ser numérica.')
    groups=[(str(name),sub[y].astype(float).values) for name,sub in d.groupby(g,dropna=True)]
    if len(groups)<2:raise ValueError('A variável de grupo precisa ter ao menos dois grupos.')
    labels=[x[0] for x in groups]; arrays=[x[1] for x in groups]
    if kind=='ttest':
        if len(groups)!=2:raise ValueError('O teste t/Welch requer exatamente dois grupos.')
        levp=stats.levene(*arrays,center='median').pvalue if min(map(len,arrays))>=2 else np.nan
        stat,p=stats.ttest_ind(*arrays,equal_var=False,nan_policy='omit')
        n1,n2=len(arrays[0]),len(arrays[1]);m1,m2=np.mean(arrays[0]),np.mean(arrays[1]);s1,s2=np.var(arrays[0],ddof=1),np.var(arrays[1],ddof=1)
        sp=math.sqrt(((n1-1)*s1+(n2-1)*s2)/(n1+n2-2)) if n1+n2>2 else np.nan;dcohen=(m1-m2)/sp if sp and np.isfinite(sp) and sp>0 else np.nan
        J=1-3/(4*(n1+n2)-9) if n1+n2>2 else 1;hedges=dcohen*J
        r=base_result('Comparação entre dois grupos','Teste t de Welch (padrão robusto a variâncias diferentes)')
        r['metrics']=[{'label':'t','value':round(float(stat),4)},{'label':'p','value':fmt_p(p)},{'label':'Hedges g','value':round(float(hedges),3) if np.isfinite(hedges) else '—'},{'label':'Levene p','value':fmt_p(levp)}]
        if np.isfinite(levp) and levp<.05:r['warnings'].append('As variâncias diferem entre grupos; foi usado Welch, que é apropriado nessa situação.')
    elif kind=='anova':
        levp=stats.levene(*arrays,center='median').pvalue if min(map(len,arrays))>=2 else np.nan
        F,p=stats.f_oneway(*arrays)
        r=base_result('Comparação de múltiplos grupos','ANOVA de uma via + Tukey HSD')
        r['metrics']=[{'label':'F','value':round(float(F),4)},{'label':'p','value':fmt_p(p)},{'label':'Grupos','value':len(groups)},{'label':'Levene p','value':fmt_p(levp)}]
        if levp<.05:r['warnings'].append('Heterogeneidade de variâncias detectada. Interprete a ANOVA clássica com cautela; considere Welch/Kruskal.')
        if len(d)>=3:
            tuk=pairwise_tukeyhsd(d[y].astype(float),d[g].astype(str));td=pd.DataFrame(tuk._results_table.data[1:],columns=tuk._results_table.data[0]);r['tables'].append(write_table(td,outdir,'anova_tukey','Pós-teste de Tukey'))
    else:
        H,p=stats.kruskal(*arrays,nan_policy='omit');r=base_result('Comparação não paramétrica','Kruskal-Wallis + comparações Mann-Whitney com Holm')
        r['metrics']=[{'label':'H','value':round(float(H),4)},{'label':'p','value':fmt_p(p)},{'label':'Grupos','value':len(groups)}]
        pairs=[]
        raw=[]
        for i in range(len(groups)):
            for j in range(i+1,len(groups)):
                u,pp=stats.mannwhitneyu(arrays[i],arrays[j],alternative='two-sided'); raw.append(pp);pairs.append([labels[i],labels[j],u,pp])
        order=np.argsort(raw);m=len(raw);adj=[None]*m;running=0
        for rank,idx in enumerate(order):
            val=min(1,raw[idx]*(m-rank));running=max(running,val);adj[idx]=running
        pdf=pd.DataFrame([{'Grupo 1':a,'Grupo 2':b,'U':u,'p':p0,'p Holm':pa} for (a,b,u,p0),pa in zip(pairs,adj)])
        r['tables'].append(write_table(pdf,outdir,'kruskal_poshoc','Comparações par-a-par'))
    summary=d.groupby(g)[y].agg(['count','mean','std','median']).reset_index();r['tables'].append(write_table(summary,outdir,'grupos_resumo','Resumo por grupo'))
    fig,ax=plt.subplots(figsize=(max(4.2,.75*len(groups)+2.2),3.4));positions=np.arange(len(groups));
    bp=ax.boxplot(arrays,positions=positions,widths=.5,patch_artist=True,showfliers=False)
    for i,box in enumerate(bp['boxes']):box.set_facecolor(PALETTE[i%len(PALETTE)]);box.set_alpha(.25);box.set_edgecolor(PALETTE[i%len(PALETTE)])
    rng=np.random.default_rng(4)
    for i,a in enumerate(arrays):
        sample=a if len(a)<=1200 else rng.choice(a,1200,replace=False);ax.scatter(rng.normal(i,.06,len(sample)),sample,s=9,alpha=.45,color=PALETTE[i%len(PALETTE)],edgecolors='none')
    ax.set_xticks(positions);ax.set_xticklabels(labels,rotation=30,ha='right');ax.set_ylabel(y);ax.set_xlabel(g);ax.spines[['top','right']].set_visible(False);ax.set_title(r['title'])
    r['figures'].append(save_figure(fig,outdir,'fig_grupos','Distribuição por grupo e observações individuais'))
    return r

def mann_kendall(x):
    a=np.asarray(x,dtype=float);a=a[np.isfinite(a)];n=len(a);S=0
    for k in range(n-1):S+=np.sign(a[k+1:]-a[k]).sum()
    unique,counts=np.unique(a,return_counts=True);varS=(n*(n-1)*(2*n+5)-np.sum(counts*(counts-1)*(2*counts+5)))/18
    if S>0:z=(S-1)/math.sqrt(varS)
    elif S<0:z=(S+1)/math.sqrt(varS)
    else:z=0
    p=2*(1-stats.norm.cdf(abs(z)));tau=S/(.5*n*(n-1)) if n>1 else np.nan
    return S,varS,z,p,tau


def analysis_timeseries(df,opt,outdir):
    t=opt.get('time');y=opt.get('y');ensure_cols(df,[t,y]);d=df[[t,y]].dropna().copy();
    if y not in numeric_cols(d):raise ValueError('A variável resposta deve ser numérica.')
    # Coerce time to datetime when possible, otherwise numeric/order.
    dt=pd.to_datetime(d[t],errors='coerce')
    if dt.notna().mean()>=.8:
        d['_t']=dt;d=d.dropna(subset=['_t']).sort_values('_t');xnum=(d['_t']-d['_t'].min()).dt.total_seconds()/86400; xlabel=t
    else:
        xnum=pd.to_numeric(d[t],errors='coerce');d=d.loc[xnum.notna()].copy();xnum=xnum.loc[xnum.notna()];d=d.iloc[np.argsort(xnum.values)];xnum=np.sort(xnum.values);xlabel=t
    yy=d[y].astype(float).values
    if len(yy)<4:raise ValueError('A série temporal precisa de pelo menos quatro observações.')
    slope,intercept,lo,hi=stats.theilslopes(yy,np.asarray(xnum,dtype=float),.95);S,varS,z,p,tau=mann_kendall(yy)
    lr=stats.linregress(np.asarray(xnum,dtype=float),yy)
    r=base_result('Tendência temporal','Mann-Kendall + inclinação de Sen + tendência linear')
    r['metrics']=[{'label':'N','value':len(yy)},{'label':'Kendall τ','value':round(float(tau),3)},{'label':'Mann-Kendall p','value':fmt_p(p)},{'label':'Sen slope','value':round(float(slope),5)},{'label':'R² linear','value':round(float(lr.rvalue**2),3)}]
    if p<.05:r['notes'].append('Há evidência de tendência monotônica pelo teste de Mann-Kendall.')
    else:r['notes'].append('Não há evidência estatística de tendência monotônica ao nível de 5%.')
    tab=pd.DataFrame([{'Mann-Kendall S':S,'z':z,'p':p,'Kendall tau':tau,'Sen slope':slope,'Sen IC95% inferior':lo,'Sen IC95% superior':hi,'Slope linear':lr.slope,'p linear':lr.pvalue,'R2 linear':lr.rvalue**2}]);r['tables'].append(write_table(tab,outdir,'tendencia_resultados','Resultados da tendência'))
    fig,ax=plt.subplots(figsize=(6.2,3.3));xplot=d['_t'] if '_t' in d else xnum;ax.plot(xplot,yy,'o-',ms=3.2,lw=.9,color=PALETTE[0],alpha=.78);xx=np.asarray(xnum,dtype=float);ax.plot(xplot,intercept+slope*xx,color=PALETTE[1],lw=1.6,label='Sen');ax.set_xlabel(xlabel);ax.set_ylabel(y);ax.spines[['top','right']].set_visible(False);ax.legend(frameon=False);ax.set_title(f'Tendência · τ={tau:.2f}, p={fmt_p(p)}')
    r['figures'].append(save_figure(fig,outdir,'fig_tendencia','Série temporal e tendência de Sen'))
    return r


def analysis_glm(df,opt,outdir):
    y=opt.get('y');xs=opt.get('variables') or ([opt.get('x')] if opt.get('x') else []);xs=[x for x in xs if x and x!=y];family=str(opt.get('family','gaussian')).lower()
    if not y or not xs:raise ValueError('Escolha resposta e preditores.')
    d=_reg_data(df,y,xs)
    fam={'gaussian':sm.families.Gaussian(),'binomial':sm.families.Binomial(),'poisson':sm.families.Poisson()}.get(family)
    if fam is None:raise ValueError('Família GLM disponível: gaussian, binomial ou poisson.')
    if family=='binomial':
        vals=sorted(d[y].unique());
        if len(vals)!=2:raise ValueError('GLM binomial requer resposta com exatamente dois valores.')
        if set(vals)!={0,1}:d[y]=d[y].map({vals[0]:0,vals[1]:1})
    X=sm.add_constant(d[xs],has_constant='add');model=sm.GLM(d[y],X,family=fam).fit()

    r['metrics']=[{'label':'N','value':int(model.nobs)},{'label':'AIC','value':round(float(model.aic),2)},{'label':'Deviance','value':round(float(model.deviance),3)},{'label':'Pearson χ²','value':round(float(model.pearson_chi2),3)}]
    ci=model.conf_int();rows=[]
    for term in model.params.index:rows.append({'Termo':term,'Estimativa':model.params[term],'Erro padrão':model.bse[term],'z':model.tvalues[term],'p':model.pvalues[term],'IC95% inferior':ci.loc[term,0],'IC95% superior':ci.loc[term,1]})
    r['tables'].append(write_table(pd.DataFrame(rows),outdir,'glm_coeficientes','Coeficientes do GLM'))
    if family=='poisson' and model.pearson_chi2/model.df_resid>1.5:r['warnings'].append('Há indício de sobredispersão no Poisson. Considere binomial negativa.')
    return r


def analysis_mixed(df,opt,outdir):

        if c not in numeric_cols(d):raise ValueError(f'Nesta versão, resposta e efeitos fixos devem ser numéricos: {c}')

    r=base_result('Modelo linear misto','MixedLM com intercepto aleatório por grupo')
    r['metrics']=[{'label':'N','value':len(dd)},{'label':'Grupos','value':dd['G'].nunique()},{'label':'LogLik','value':round(float(model.llf),2)},{'label':'Convergiu','value':'Sim' if model.converged else 'Não'}]
    ci=model.conf_int();rows=[]

    r['tables'].append(write_table(pd.DataFrame(rows),outdir,'misto_coeficientes','Coeficientes do modelo misto'))
    if not model.converged:r['warnings'].append('O modelo não convergiu; simplifique a estrutura ou revise escala/colinearidade.')
    return r
